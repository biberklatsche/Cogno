#!/bin/zsh
# Cogno Integration Script for Zsh
# Provides OSC markers, prompt hooks, and shell integration features

if [[ -n "${COGNO_ZSH_INTEGRATION_LOADED}" ]]; then
  return 0
fi
typeset -g COGNO_ZSH_INTEGRATION_LOADED=1

setopt PROMPT_SUBST
autoload -Uz add-zsh-hook

# State
typeset -g COGNO_COUNT=0
typeset -g COGNO_LAST_CMD=""
typeset -g COGNO_CMD_SEEN=0

# Extract command name: trim, drop leading KEY=VALUE prefixes, take first token
_cogno_extract_command_name() {
  local line="$1"
  line="${line#${line%%[![:space:]]*}}"

  local assignment_re='^[A-Za-z_][A-Za-z0-9_]*=[^[:space:]]+[[:space:]]+'
  while [[ "$line" =~ $assignment_re ]]; do
    line="${line#${MATCH}}"
  done

  line="${line#${line%%[![:space:]]*}}"
  if [[ -z "$line" ]]; then
    print -r -- ""
    return
  fi

  print -r -- "${line%%[[:space:]]*}"
}

_cogno_preexec() {
  # $1 = command line as entered
  COGNO_LAST_CMD="$1"
  COGNO_CMD_SEEN=1
}

_cogno_precmd() {
  local last_ec=$?

  ((COGNO_COUNT++))

  local ts="$COGNO_COUNT"

  local host="${HOST%%.*}"
  local cwd="$PWD"

  local cmd=""
  local command_exists="false"
  if (( COGNO_CMD_SEEN )); then
    cmd="$COGNO_LAST_CMD"
    local command_name
    command_name="$(_cogno_extract_command_name "$COGNO_LAST_CMD")"
    if [[ -n "$command_name" ]] && whence -w -- "$command_name" >/dev/null 2>&1; then
      command_exists="true"
    fi
  fi

  # Sanitize command for OSC payload
  cmd="${cmd//$'\r'/}"
  cmd="${cmd//$'\n'/\\n}"
  cmd="${cmd//$'\e'/}"
  cmd="${cmd//$'\a'/}"
  cmd="${cmd//;/\\;}"
  cmd="${cmd//|/\\|}"

  # Compute and insert commandExists into the existing OSC payload
  local osc=$'\e]733;COGNO:PROMPT;'"returnCode=${last_ec};user=${USER};machine=${host};directory=${cwd};id=${ts};command=${cmd};commandExists=${command_exists};"$'\e\\'

  print -n -r -- "$osc"

  # Reset for next prompt
  COGNO_CMD_SEEN=0
  COGNO_LAST_CMD=""

  # Re-assert the Cogno prompt every cycle: prompt frameworks loaded from the
  # user's rc files (powerlevel10k, starship, oh-my-posh, ...) rewrite PROMPT
  # from their own precmd hooks and would otherwise remove the hidden marker.
  PROMPT="${COGNO_PROMPT}"
  RPROMPT=""

  return $last_ec
}

add-zsh-hook preexec _cogno_preexec
add-zsh-hook precmd  _cogno_precmd

# Keep Cogno's precmd last so it runs after any prompt-framework hooks that
# were registered by the user's rc files.
precmd_functions=(${precmd_functions:#_cogno_precmd} _cogno_precmd)

typeset -g COGNO_PROMPT=$'\n%{\e[8m%}^^#${COGNO_COUNT}%{\e[0m%}\n'
PROMPT="${COGNO_PROMPT}"
RPROMPT=""

# Native line-editor channel: the app writes replace requests into a
# per-session FIFO; `zle -F` invokes the handler only while ZLE is reading at
# the prompt, so a request can never leak into a running program's stdin.
# Message format (written by the Rust backend, one request per line):
#   action;cursorIndex;autoExecute;text
# The text is the last field (may contain raw semicolons); backslash, CR and
# LF arrive escaped and are decoded with ${(g::)...}.

# Registered with `zle -F -w`, so the handler runs as a widget: only then may
# it edit BUFFER/CURSOR and call accept-line. A plain `zle -F` handler has no
# widget facilities and its buffer changes are silently lost. Display updates
# are not automatic at this level - `zle -R` forces the redisplay.
_cogno_le_handle() {
  local fd="$1" line="" action="" cursor="" autoexec="" text=""
  if ! IFS= read -r -u "$fd" line; then
    # Broken channel: unregister so ZLE does not spin on a dead fd.
    zle -F "$fd" 2>/dev/null
    return 0
  fi
  IFS=';' read -r action cursor autoexec text <<< "$line"
  case "$action" in
    replaceCurrentInput)
      BUFFER="${(g::)text}"
      if [[ "$cursor" == <-> ]] && (( cursor <= ${#BUFFER} )); then
        CURSOR="$cursor"
      else
        CURSOR="${#BUFFER}"
      fi
      # The autoexec field is deliberately NOT handled here: accept-line (and
      # zle -U) are no-ops inside fd-handler widgets. The backend submits by
      # writing CR to the PTY right after this message - ZLE services fd
      # handlers before pending keyboard bytes, so that CR always accepts the
      # freshly replaced buffer. Accepting here too would double-submit.
      zle -R
      ;;
  esac
  return 0
}

_cogno_le_setup() {
  [[ -n "${COGNO_LINE_EDITOR_PIPE}" && -p "${COGNO_LINE_EDITOR_PIPE}" ]] || return 1
  zle -N _cogno_le_handle 2>/dev/null || return 1
  # Open read-write: the read end never sees EOF when the app closes its
  # write end after each request, and the open cannot block.
  typeset -gi COGNO_LE_FD=-1
  if ! exec {COGNO_LE_FD}<>"${COGNO_LINE_EDITOR_PIPE}"; then
    return 1
  fi
  # -w requires zsh >= 5.2; on failure the session simply reports no native
  # actions and the app stays on the raw fallback.
  if ! zle -F -w "${COGNO_LE_FD}" _cogno_le_handle 2>/dev/null; then
    exec {COGNO_LE_FD}>&-
    return 1
  fi
  return 0
}

# Capability handshake: report what this session's integration can actually
# do. The app gates its native line-editor paths on this report and never
# assumes a capability statically - native actions are only announced when
# the FIFO channel actually came up.
autoload -Uz is-at-least
_cogno_bracketed_paste="false"
if is-at-least 5.1 "${ZSH_VERSION}"; then
  _cogno_bracketed_paste="true"
fi
_cogno_native_actions=""
if _cogno_le_setup; then
  _cogno_native_actions="replaceCurrentInput"
fi
print -n -r -- $'\e]733;COGNO:CAPS;shell=zsh;shellVersion='"${ZSH_VERSION}"$';nativeActions='"${_cogno_native_actions}"$';bracketedPaste='"${_cogno_bracketed_paste}"$';\e\\'
unset _cogno_bracketed_paste _cogno_native_actions
